cmake_minimum_required(VERSION 3.18)
project(yolo_engine LANGUAGES CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ENGINE_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../artifacts/engine/generated" CACHE PATH "Generated model directory")
get_filename_component(ENGINE_GENERATED_DIR "${ENGINE_GENERATED_DIR}" ABSOLUTE)
add_compile_options(-O3 -march=x86-64-v3 -fno-math-errno -Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)
add_library(engine_runtime runtime/runtime.cpp runtime/avx2.cpp runtime/packed_conv.cpp runtime/packed_conv_int8.cpp runtime/transpose.cpp runtime/depthwise.cpp)
set_source_files_properties(runtime/packed_conv_int8.cpp PROPERTIES COMPILE_OPTIONS "-mavxvnni;-ffp-contract=off")
target_include_directories(engine_runtime PUBLIC runtime)
target_link_libraries(engine_runtime PUBLIC Threads::Threads)
set_source_files_properties("${ENGINE_GENERATED_DIR}/weights.S" PROPERTIES COMPILE_OPTIONS "-Wa,-I${ENGINE_GENERATED_DIR}" OBJECT_DEPENDS "${ENGINE_GENERATED_DIR}/weights.bin")
add_executable(yolo_engine cpp/main.cpp "${ENGINE_GENERATED_DIR}/model.cpp" "${ENGINE_GENERATED_DIR}/weights.S")
target_include_directories(yolo_engine PRIVATE "${ENGINE_GENERATED_DIR}")
target_include_directories(yolo_engine SYSTEM PRIVATE ../native)
target_link_libraries(yolo_engine PRIVATE engine_runtime)
enable_testing()
add_executable(test_kernels tests/test_kernels.cpp)
target_link_libraries(test_kernels PRIVATE engine_runtime)
add_test(NAME kernels COMMAND test_kernels)
add_test(NAME kernels_threads COMMAND test_kernels 6)
add_executable(test_int8 tests/test_int8.cpp)
target_link_libraries(test_int8 PRIVATE engine_runtime)
add_test(NAME int8 COMMAND test_int8)
add_test(NAME int8_threads COMMAND test_int8 6)
add_executable(test_thread_pool tests/test_thread_pool.cpp)
target_link_libraries(test_thread_pool PRIVATE engine_runtime)
add_test(NAME thread_pool COMMAND test_thread_pool)
option(ENGINE_TSAN "Instrument runtime and tests with ThreadSanitizer" OFF)
if(ENGINE_TSAN)
    target_compile_options(engine_runtime PUBLIC -fsanitize=thread -g)
    target_link_options(engine_runtime PUBLIC -fsanitize=thread)
endif()
