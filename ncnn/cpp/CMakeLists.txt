cmake_minimum_required(VERSION 3.24)
project(yolo_ncnn LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ncnn REQUIRED CONFIG PATHS /usr/lib/cmake/ncnn)
add_executable(yolo_ncnn main.cpp)
target_include_directories(yolo_ncnn PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../native")
target_compile_options(yolo_ncnn PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(yolo_ncnn PRIVATE ncnn ${CMAKE_DL_LIBS})
